# The static CyberShuttle Jupyter site and, on its own name, the csctl API it calls.
server {
    listen 443 ssl;
    listen [::]:443 ssl ipv6only=on;
    server_name jupyter.cybershuttle.org;
    ssl_certificate /etc/letsencrypt/live/jupyter.cybershuttle.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/jupyter.cybershuttle.org/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    root /var/www/jupyter.cybershuttle.org;
    index index.html;
    location / {
        try_files $uri $uri/ =404;
    }
}

server {
    listen 443 ssl;
    listen [::]:443 ssl;
    server_name jupyterapi.cybershuttle.org;
    ssl_certificate /etc/letsencrypt/live/jupyter.cybershuttle.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/jupyter.cybershuttle.org/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    # The SSH-auth WebSocket carries the bearer in Sec-WebSocket-Protocol, past nginx's 8k default.
    large_client_header_buffers 4 64k;
    location / {
        proxy_pass http://127.0.0.1:8045;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection $connection_upgrade;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 3600s;
        proxy_send_timeout 3600s;
        proxy_buffering off;
    }
}

server {
    listen 80;
    listen [::]:80;
    server_name jupyter.cybershuttle.org jupyterapi.cybershuttle.org;
    return 301 https://$host$request_uri;
}
