# The Custos portal, plus the one Custos core endpoint csctl calls to resolve a caller.
server {
    listen 443 ssl;
    listen [::]:443 ssl;
    server_name custos.cybershuttle.org;
    ssl_certificate /etc/letsencrypt/live/custos.cybershuttle.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/custos.cybershuttle.org/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    location = /me {
        proxy_pass http://127.0.0.1:8100;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
    location / {
        proxy_pass http://127.0.0.1:3100;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}

server {
    listen 80;
    listen [::]:80;
    server_name custos.cybershuttle.org;
    return 301 https://$host$request_uri;
}
